% 4강 수식
%% E01-지연
T_d \approx \underbrace{T_s}_{\text{계산}} + \underbrace{0.5\,T_s}_{\text{PWM 홀드}} = 1.5\,T_s = 150\ \mu\mathrm{s}
%% E02-위상
\phi_d = \omega\,T_d = 360^\circ\cdot f\,T_d
%% E03-ZOH
G_{zoh}(s) = \frac{1 - e^{-sT_s}}{sT_s}\;\Rightarrow\; G_{zoh}(j\omega) = e^{-j\omega T_s/2}\,\frac{\sin(\omega T_s/2)}{\omega T_s/2}
%% E04-출력각
v^*_{\alpha\beta} = v^*_{dq}\,e^{\,j(\theta_k + 1.5\,\omega T_s)}
%% E05-톱니
v_q(t) = -\sin\big(\omega(t - t_k)\big) \approx -\omega(t - t_k),\qquad \overline{v_q} = -\frac{\omega T_s}{2}
%% E06-오프셋
\Delta\omega_{pll}(t_k) = k_p\,\frac{\omega T_s}{2}\;\Rightarrow\;\Delta f \approx 1.18\times\frac{0.0377}{2}\times 60 \approx 1.33\ \mathrm{Hz}
%% E07-1차근사
\frac{1}{1 + sT_d}:\quad \angle = -\arctan(\omega T_d) = -37^\circ\ \text{at}\ 800\ \mathrm{Hz}
%% E08-예제
\phi_d = 360^\circ\times 1000 \times 150\times 10^{-6} = 54^\circ
%% E09-예제PLL
k_p = \frac{2\zeta\,\omega_n}{\omega_b} = \frac{2\times 0.707\times 2\pi\times 20}{377} \approx 0.471,\qquad \Delta f \approx 0.471\times\frac{0.0377}{2}\times 60 \approx 0.53\ \mathrm{Hz}

def scr_step(mode, f_pll, scr_after):
    p = Params(converter="average", scr=5.0, P_ref=0.8,
               f_pll=f_pll, gfl_frt_enter=0.5) # FRT 순환 배제
    sc = Scenario(t_end=1.0, scr=lambda t: 5.0 if t < 0.2 else scr_after)
    r = simulate(mode, p, sc, dt=1e-5, x0=steady_state(mode, p, dt=1e-5))
    return oscillation_check(r["t"], r["P"], 0.4) # 5~300 Hz 진동, 저주파 스윙

for mode, f_pll, scr in [("GFL",20,1.5), ("GFL",50,1.5), ("GFL",50,2.0),

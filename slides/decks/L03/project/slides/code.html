모델 파라미터로 공진과 리플 상한을 확인한다

import sys; sys.path.insert(0, "simulation/emt")
import numpy as np
from gfm_gfl_emt import Params
p = Params()
for Lg in (p.Lt, p.Lt + 1/3, p.Lt + 1/2):
    f_res = p.f0*np.sqrt((p.Lf + Lg)/(p.Lf*Lg*p.Cf))
    print(f"Lg={Lg:.3f} -> {f_res:.0f} Hz")
vdc = p.V_dc/p.V_base_peak # 2.256 pu
L = p.Lf/p.wb # pu → '초'
print(vdc/(4*L*p.f_sw))

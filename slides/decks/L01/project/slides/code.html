import numpy as np
wb = 2*np.pi*60.0
t = np.arange(0, 0.05, 1e-5)
th = wb*t # Park 각도 θ = ωt
a = np.exp(2j*np.pi/3)
ia = np.cos(wb*t - np.pi/6) # 30° 늦은 전류
ib = np.cos(wb*t - np.pi/6 - 2*np.pi/3)
ic = np.cos(wb*t - np.pi/6 + 2*np.pi/3)
i_sv = 2/3*(ia + a*ib + a**2*ic) # 공간벡터
i_dq = i_sv*np.exp(-1j*th) # Park
print(i_dq.real.mean(), i_dq.imag.mean())

출력

0.866 −0.5
